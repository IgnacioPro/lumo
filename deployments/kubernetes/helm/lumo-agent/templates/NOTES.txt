  kubectl logs -n {{ include "lumo-agent.namespace" . }} -l app.kubernetes.io/component=cluster-monitor --tail=50
{{- end }}

# Check agent health
kubectl get pods -n {{ include "lumo-agent.namespace" . }} -o wide

# Access health endpoint
kubectl port-forward -n {{ include "lumo-agent.namespace" . }} service/lumo-agent-cluster 8080:8080
curl http://localhost:8080/health

# Access Prometheus metrics
kubectl port-forward -n {{ include "lumo-agent.namespace" . }} service/lumo-agent-cluster 9090:9090
curl http://localhost:9090/metrics
